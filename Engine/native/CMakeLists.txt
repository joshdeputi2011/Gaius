cmake_minimum_required(VERSION 3.16)
project(gaius_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(gaius_engine SHARED src/gaius.cpp)
target_include_directories(gaius_engine PUBLIC include)
target_compile_definitions(gaius_engine PRIVATE GAIUS_ENGINE_BUILD PUBLIC GAIUS_ENGINE_SHARED)
if(NOT ANDROID AND NOT IOS)
  enable_testing()
  add_executable(gaius_engine_tests tests/test_engine.cpp)
  target_link_libraries(gaius_engine_tests PRIVATE gaius_engine)
  add_test(NAME engine COMMAND gaius_engine_tests)
endif()
